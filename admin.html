Firebase 接続状態

Stripe 設定
firebase-config.js の stripeKey を本番キーに変更することで、本番環境の決済が有効になります。
Firestore セキュリティルール（推奨設定）
Firebase コンソール → Firestore → ルールに以下を設定してください
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // 商品: 全員読み取り可、管理者のみ書き込み可
    match /products/{id} {
      allow read: if true;
      allow write: if request.auth != null &&
        exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }
    // 注文: 本人と管理者のみ
    match /orders/{id} {
      allow read, write: if request.auth != null;
    }
    // 管理者リスト: 管理者のみ
    match /admins/{uid} {
      allow read, write: if request.auth != null &&
        exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }
  }
}
